SQL Query Sinh Tự Động (Dialect: Oracle) ORACLE AST 
SELECT u.NAME AS "TÊN TỈNH/THÀNH",
       SUM(cr.COMMISSION_AMOUNT) AS "TỔNG HOA HỒNG",
       COUNT(u.CODE) AS "SỐ ĐƠN VỊ"
FROM CFG_BHXH_UNITS u
LEFT JOIN CFG_COMMISSION_RATES cr
  ON u.CODE = cr.UNIT_CODE
WHERE u.STATUS = 1
  AND (:provinceCode IS NULL OR u.PROVINCE_CODE = :provinceCode)
  AND cr.TRANS_DATE BETWEEN TO_DATE(:fromDate, 'DD/MM/YYYY') 
                        AND TO_DATE(:toDate, 'DD/MM/YYYY')
GROUP BY u.NAME
ORDER BY 2 DESC;
